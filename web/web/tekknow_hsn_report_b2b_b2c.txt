select hsn_code,sum(pcs),sgst_per+cgst_per+igst_per as gst_per,ifnull(sum(taxable_amt),0) as taxable_amt,
ifnull(sum(taxable_amt*f1.cgst_per/100),0) as cgst_amt,
ifnull(sum(taxable_amt*f1.sgst_per/100),0) as sgst_amt,
ifnull(sum(taxable_amt*f1.igst_per/100),0) as igst_amt ,
l1.gstin_no

from f_invoice f1,f_invoice_det f2,ledger_mast l1 where 
f1.ledger_code=l1.ledger_code and 
f1.invoice_code=f2.invoice_code and bill_date between '2025/06/01' and '2025/06/30' and ifnull(status,'N')<>'C' group by hsn_code,l1.gstin_no ,sgst_per,cgst_per,igst_per having 
length(ifnull(l1.gstin_no,'') )=0   
;
