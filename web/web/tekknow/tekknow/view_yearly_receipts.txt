alter view v_yearly_receipts as select mth_code,c_code, mth_name, sum( case when rece_date>= fdate.from_date and rece_date<=fdate.to_date then amount else 0 end) as cur_sales, sum( case when rece_date>= fdate.from_date1 and rece_date<=fdate.to_date1 then amount else 0 end) as prev_sales,sum( case when rece_date>= fdate.from_date2 and rece_date<=fdate.to_date2 then amount else 0 end) as bef_prev_sales from rece_mast,mth1,fdate,fin_date where (fdate.from_date between fin_date.from_date and fin_date.to_date) and month(rece_mast.rece_date)=mth1.mth_no and ifnull(status,'N')<>'C'  group by c_code,mth_name,mth_code
