alter view   v_payable  as

select bill_date as dt,bill_date,bill_no,ledger_code, net_amount as total_amt,0 as paid_amount,0 as discount,c_code from purc_sale p1, purc_sale_det p2 where p1.tran_code=p2.tran_code and tran_type in('P','O') and  ifnull(status,'N')<>'C' and ledger_code<>1 and bill_date>=date('2019-04-01') 

union all 

select party_inv_date as dt,party_inv_date ,party_inv_no,ledger_code, ifnull(total_amount,0)-ifnull((select ifnull(total_amt,0) from purc_ret p1 where p1.against_invoice_code=p1.inv_code and ifnull(status,'N')<>'C'),0) - ifnull((select ifnull(sum(adj_amount),0) from advance_mapping_det a1,advance_mapping a2 where a1.mapping_code=a2.mapping_code and ifnull(status,'N')<>'C' and a2.advance_type='I' and a1.invoice_code=p1.inv_code and a2.c_code=p1.c_code),0) as total_amt,0 as paid_amount,0 as discount,c_code from purc_inv p1 where  ifnull(status,'N')<>'C' and ledger_code<>1 and party_inv_date>=date('2019-04-01') and trim(party_inv_no)<>'' and transaction_type='I' 

union all 

select   inv_date as dt, inv_date , inv_no,ledger_code, ifnull(total_amount,0)-ifnull((select ifnull(total_amt,0) from purc_ret p1 where p1.against_invoice_code=p1.inv_code and ifnull(status,'N')<>'C'),0) - ifnull((select ifnull(sum(adj_amount),0) from advance_mapping_det a1,advance_mapping a2 where a1.mapping_code=a2.mapping_code and ifnull(status,'N')<>'C' and a2.advance_type='I' and a1.invoice_code=p1.inv_code and a2.c_code=p1.c_code),0) ,0 as paid_amount,0 as discount,c_code from purc_inv p1 where  ledger_code<>1 and  inv_date>=date('2019-04-01') and transaction_type='B' and ifnull(status,'N')<>'C'

union all 

select pay_date,bill_date,bill_no,r2.party_code,0,paid_amount,discount,c_code from pay_det r1,pay_mast r2 where r1.pay_code=r2.pay_code and ifnull(status,'N')<>'C'  and pay_date>=date('2019-04-01') and tran_type in ('P','O','Y','B')

union all 
 
select bill_date,bill_date,bill_no,ledger_code,0,net_amount,0,c_code from purc_sale p1, purc_sale_det p2 where p1.tran_code=p2.tran_code and tran_type='D' and  ifnull(status,'N')<>'C' and ledger_code<>1 and bill_date>=('2019-04-01') 

union all 


select bill_date as dt,bill_date,bill_no,ledger_code, net_amount as total_amt,0 as paid_amount,0 as discount,c_code from purc_sale p1, purc_sale_det p2 where p1.tran_code=p2.tran_code and tran_type in('Y') and  ifnull(status,'N')<>'C' and ledger_code<>1  

union all 

select advance_date,advance_date,advance_no,case when main_ledger_code is null or main_ledger_code=0 then party_ledger_code else main_ledger_code end,amount-(select ifnull(sum(adj_amount),0) from advance_mapping_det a1,advance_mapping a2 where a1.mapping_code=a2.mapping_code and a1.advance_code=advance.advance_code and ifnull(a2.status,'N')<>'C'),0,0,c_code from advance,ledger_mast l1 where advance.party_ledger_code=l1.ledger_code and ifnull(status,'N')<>'C' and advance.advance_type='I'
