select d1.design_code product_code,prod_code,product_name,d1.hsn_code,rate,sum(pcs) as qty,sum(mtrs) mtrs,d1.unit_code,unit_name,sum(amount) amount,sum(disc_amt) disc_amt,sum(taxable_amt) taxable_amt,sum(freight_amt) freight_amt,d1.invoice_code dc_code from dc_det d1,product_mast p1,unit_mast u1 where d1.design_code=p1.product_code and d1.unit_code=u1.unit_code and d1.invoice_code in $(bncode) group by d1.design_code,product_code,prod_code,product_name,d1.hsn_code,rate
